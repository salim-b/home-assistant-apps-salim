#include <tunables/global>

profile hickory-dns flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  ## Mediation mechanics and gotchas: repository-root KNOWLEDGE.md §"AppArmor: mediation and profile mechanics";
  ## app facts and the full profile surface: hickory-dns/KNOWLEDGE.md §"AppArmor profile surface".

  # Generic file access (`file,` per the upstream HA convention – the confinement value sits in the non-file classes,
  # repository-root KNOWLEDGE.md §"AppArmor: mediation and profile mechanics"). No own signal rules: the app runs one
  # profile label in its own PID namespace (no `host_pid`, no sub-profiles), so every reachable signal target is
  # same-label, and <abstractions/base> (AppArmor 3.1.7) grants the rest – same-label delivery (`signal
  # peer=@{profile_name}`: s6-supervise signals the daemon on stop, shell job control), existence probes
  # (`signal (receive, send) set=("exists")`) and receive from the unconfined host runtime (`signal (receive)
  # peer=unconfined`: how docker stop/restart reach PID 1). The upstream template's blanket `signal,` adds only
  # cross-label grants nothing here can exercise.
  file,

  # The never-root launcher's grants - which syscall needs which capability, the capable()-hook mediation and the
  # one-capability end state: hickory-dns/KNOWLEDGE.md §"Port 53 binding and privileges".
  capability net_bind_service,
  capability setgid,
  capability setuid,
  capability setpcap,

  # DNS both ways (:53 listeners v4/v6, encrypted upstream connections) and bashio's supervisor-API HTTP -
  # enumerated surface: hickory-dns/KNOWLEDGE.md §"AppArmor profile surface".
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,

  # S6-Overlay
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Options file and the app's persistent data dir (generated named.toml, upstream-resolution cache)
  /data/** rw,

  # AppArmor status reporting (s6 sets this up; harmless otherwise)
  /sys/kernel/security/apparmor/** r,

  # App config surface (generated named.toml + zones; hickory-dns/KNOWLEDGE.md §"Config-file and data layout").
  # Blanket-covered; documents what the scripts read and write.
  /etc/hickory-dns/{,**} rw,

  # The s6 service command line the run script reads (hickory-dns/KNOWLEDGE.md §"Config-file and data layout").
  # Blanket-covered; documents what is read.
  /etc/hickory-dns.args r,

  # Hostname-upstream resolution (resolve --system) and the identity lookups (hickory-dns/KNOWLEDGE.md
  # §"Port 53 binding and privileges"). Blanket-covered; enumerated as documentation.
  /etc/passwd r,
  /etc/group r,
  /etc/resolv.conf r,
  # System trust store read by rustls-platform-verifier when connecting to DoT/DoH/DoQ upstreams
  # (ca-certificates-bundle). Blanket-covered documentation rule.
  /etc/ssl/certs/{,**} r,

  # Runtime check scripts
  /etc/cont-init.d/config.sh rix,
}
