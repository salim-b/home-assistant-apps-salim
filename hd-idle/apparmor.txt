#include <tunables/global>

profile hd-idle flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  ## Conventions & references: modeled on the nfs-server app profile and the
  ## official dev-docs AppArmor template; the AppArmor mediation gotchas
  ## documented there apply here too (capabilities are mediated, network
  ## mediation is opt-in per class, HAOS produces no AppArmor audit logs).
  ## See hd-idle/KNOWLEDGE.md and the repository-root KNOWLEDGE.md.

  # Generic file access, signals (s6 supervision, bash job control).
  # `file,` is the upstream HA convention (docs template + dnsmasq + the
  # nfs-server app): tightening file rules is impractical because HAOS
  # produces no AppArmor audit/denial logs - silent breakage risk. The
  # profile's confinement value sits in the non-file classes (network,
  # capability).
  file,
  signal,

  # CAP_SYS_RAWIO for SCSI passthrough: the kernel only allows non-read-
  # safe SCSI commands (ATA PASS-THROUGH - the `ata` spindown path and
  # smartctl's ATA commands) when capable(CAP_SYS_RAWIO) holds
  # (drivers/scsi/scsi_ioctl.c scsi_cmd_allowed(); read-safe commands like
  # START STOP UNIT - the `scsi` path - need no capability). AppArmor
  # MEDIATES THAT CHECK: the security_capable LSM hook runs
  # apparmor_capable() after the credential check, so a profile without a
  # `capability` rule denies the check itself - the container may carry the
  # cap (verified via CapEff), but capable() still returns false and the
  # kernel falls through to `return open_for_write` (EPERM for hd-idle's
  # O_RDONLY fds). Before this rule every ata-path SG_IO failed with
  # "operation not permitted" while the `scsi` path worked via the kernel's
  # read-safe list; HAOS logs no AppArmor denials, so the root cause
  # surfaced only through source analysis.
  capability sys_rawio,

  # Network: mediation is enabled by the PRESENCE of any network rule. The
  # app only needs supervisor-API HTTP (bashio) and DNS resolution; the
  # hd-idle daemon itself makes no sockets (its sub-profile denies network
  # entirely).
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,

  # S6-Overlay
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within the app's data dir
  /data/** rw,

  # AppArmor status reporting (s6 sets this up; harmless otherwise)
  /sys/kernel/security/apparmor/** r,

  # Disk introspection and spindown targets. hd-idle opens disk nodes
  # O_RDONLY and issues SG_IO ioctls (AppArmor 3.x mediates ioctls via the
  # file's r/w permission). /dev/disk/by-* is read for device-path
  # resolution (config.sh), /sys/class/block and /sys/block for disk
  # attribution (partition/device-mapper stats) and USB-bridge detection
  # (sgio's idVendor walk). The explicit rules document the app's disk
  # access surface; the supervisor's cgroup device rules (granted per
  # configured drive via the device-type option fields) are the layer that
  # actually scopes WHICH disks are reachable, and it refuses devices the
  # system uses. (The nfs-server KNOWLEDGE.md AppArmor notes apply.)
  /dev/sd* rwk,
  /dev/disk/{,**} r,
  /sys/class/block/{,**} r,
  /sys/block/{,**} r,

  # Generated hd-idle command line (cont-init config.sh -> s6 run script)
  /etc/hd-idle.args rw,

  # Runtime check scripts
  /etc/cont-init.d/config.sh rix,
  /etc/s6-overlay/scripts/* rix,

  ## The hd-idle daemon runs under its own tighter sub-profile (cx = change
  ## profile on execute): no networking at all, and only the disk/sysfs
  ## surface it actually needs.
  /usr/sbin/hd-idle cx -> hdidle,

  profile hdidle flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    file,
    # CAP_SYS_RAWIO for the daemon's ata path: see the top profile
    # (AppArmor mediates the capable() check the kernel's SG_IO permission
    # gate relies on - without this rule every ATA PASS-THROUGH fails with
    # EPERM even though the container carries the capability).
    capability sys_rawio,
    signal (send) set=(kill,term,int,hup,cont),
    # Official docs template: scope receive to the app's own top profile
    # (the supervisor renames it to the installed slug on the device, e.g.
    # local_hd-idle - the *_hd-idle glob covers repo installs too).
    signal (receive) peer=*_hd-idle,

    # hd-idle creates no sockets: reads /proc/diskstats, sysfs, and issues
    # SG_IO ioctls on the disk nodes. A deny-only network class denies
    # everything in it.
    deny network,

    /usr/sbin/hd-idle rm,
    /lib/** mr,
    /dev/sd* rwk,
    /dev/disk/{,**} r,
    /proc/diskstats r,
    /sys/class/block/{,**} r,
    /sys/block/{,**} r,
    # Activity log (-l, written only on spinup/spindown events)
    /data/** rw,
    /proc/[0-9]*/{,cmdline,mounts,stat} r,
    /tmp/** rwk,
  }
}
