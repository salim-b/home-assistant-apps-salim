#include <tunables/global>

profile hd-idle flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  ## Modeled on the nfs-server profile and the official dev-docs AppArmor template. Mediation mechanics and gotchas:
  ## repository-root KNOWLEDGE.md §"AppArmor: mediation and profile mechanics"; app facts: hd-idle/KNOWLEDGE.md.

  # Generic file access, signals (s6 supervision, bash job control). `file,` is the upstream HA convention - the
  # confinement value sits in the non-file classes (network, capability); repository-root
  # KNOWLEDGE.md §"AppArmor: mediation and profile mechanics".
  file,
  signal,

  # Non-read-safe SCSI passthrough (hd-idle's `ata` command type, smartctl) requires CAP_SYS_RAWIO, and AppArmor
  # mediates capable() checks itself - without this rule every ata-path SG_IO fails with EPERM:
  # hd-idle/KNOWLEDGE.md §"SG_IO privileges".
  capability sys_rawio,

  # Mediation is enabled by the PRESENCE of any network rule; the app needs supervisor-API HTTP (bashio) and DNS
  # only - the hd-idle daemon makes no sockets (its sub-profile denies network entirely).
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,

  # S6-Overlay
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within the app's data dir
  /data/** rw,

  # AppArmor status reporting (s6 sets this up; harmless otherwise)
  /sys/kernel/security/apparmor/** r,

  # Disk introspection and spindown targets (O_RDONLY + SG_IO; mediated via the file's r/w permission). /dev/disk/by-*
  # for path resolution (config.sh), /sys/block for attribution and USB-bridge detection. The supervisor's cgroup
  # device rules (granted per configured drive via the device-type option fields) actually scope WHICH disks are
  # reachable, and refuse system-used devices.
  /dev/sd* rwk,
  /dev/disk/{,**} r,
  /sys/class/block/{,**} r,
  /sys/block/{,**} r,

  # Generated-config file surface (not load-bearing under the `file,` blanket; documents what the scripts read and
  # write): args (cont-init -> run script), drives (hd-idle-pmfix), resolution baseline (genargs --check), and the
  # previous-drives copy hd-idle-pmfix --since diffs against.
  /etc/hd-idle.args rw,
  /etc/hd-idle.drives rw,
  /etc/hd-idle.drives.prev rw,
  /etc/hd-idle.resolution rw,

  # Runtime check scripts
  /etc/cont-init.d/config.sh rix,

  # Drive-PM treatment (hd-idle-pmfix): smartctl issues ATA SET FEATURES (non-read-safe, hence the capability rule
  # above); sysfs coverage broad per the header note.
  /usr/sbin/smartctl rix,
  /sys/devices/{,**} r,

  ## The hd-idle daemon runs under a tighter sub-profile (cx): no networking, only the disk/sysfs surface it needs.
  /usr/sbin/hd-idle cx -> hdidle,

  profile hdidle flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    file,
    # CAP_SYS_RAWIO for the daemon's ata path: see the top profile (hd-idle/KNOWLEDGE.md §"SG_IO privileges").
    capability sys_rawio,
    signal (send) set=(kill,term,int,hup,cont),
    # Official docs template: scope receive to the app's own top profile (the supervisor renames it to the installed
    # slug on the device - the glob covers repo installs too).
    signal (receive) peer=*_hd-idle,

    # hd-idle creates no sockets (reads /proc/diskstats, sysfs, SG_IO ioctls); a deny-only network class denies everything in it.
    deny network,

    /usr/sbin/hd-idle rm,
    /lib/** mr,
    /dev/sd* rwk,
    /dev/disk/{,**} r,
    /proc/diskstats r,
    /sys/class/block/{,**} r,
    /sys/block/{,**} r,
    # Activity log (-l, written only on spinup/spindown events)
    /data/** rw,
    /proc/[0-9]*/{,cmdline,mounts,stat} r,
    /tmp/** rwk,
  }
}
