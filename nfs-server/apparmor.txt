#include <tunables/global>

profile nfs-server flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  ## Conventions & references (device-enforcement-verified on HAOS 18.3 / OS Agent 1.14 / AppArmor 3.1.7):
  ## shape modeled on the official dev-docs AppArmor template and the shipped dnsmasq app profile; sub-profile
  ## signal receive is peer-scoped per the docs template (the unscoped send grants there are template-inherited
  ## headroom - nothing in them sends signals; nfs-server/TODO.md tracks scoping/dropping them); `file,` blanket per
  ## upstream convention - HAOS produces
  ## no AppArmor audit/denial logs, so tightening file rules risks silent breakage (the confinement value sits in
  ## the non-file classes: mount/capability). Facts:
  ## nfs-server/KNOWLEDGE.md §"AppArmor profile — app-specific grants";
  ## root KNOWLEDGE.md §"AppArmor: mediation and profile mechanics"; procedure: .agents/skills/ha-app-apparmor.

  # Generic file access, signals (s6 supervision, bash job control). Bare `signal,` matches the upstream dnsmasq
  # app profile convention: it grants send AND receive for all signals/peers (cross-label delivery needs rules on
  # both sides).
  file,
  signal,

  # Network mediation is enabled by the PRESENCE of any network rule; once enabled, anything not explicitly
  # allowed is denied - a close allowlist: inet/inet6 stream+dgram (bashio supervisor API, DNS, the Docker
  # HEALTHCHECK's /dev/tcp probe) and netlink raw (busybox `ip` address detection in nfsd-start/nfsd-healthcheck).
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  network netlink raw,

  # S6-Overlay
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within the app's data dir
  /data/** rw,

  # AppArmor status reporting (s6 sets this up; harmless otherwise)
  /sys/kernel/security/apparmor/** r,

  # Address detection (APP_IP): `ip` is a busybox applet, hence the
  # /bin/busybox ix (see KNOWLEDGE.md on busybox-applet exec mediation)
  /sbin/ip rix,
  /bin/busybox ix,

  # Mount the nfsd filesystem (SYS_ADMIN) - the app's core operation. busybox mount passes the device string
  # ("nfsd"/"sunrpc") as the mount source, so sources cannot be constrained to a path - scope by filesystem type
  # instead. The per-share mirrors are mount --bind (fstype = the underlying fs), umounted again by the down
  # script. AppArmor also mediates capable(CAP_SYS_ADMIN) during mount(2) even when the `mount` rule itself
  # grants it (plain Docker containers carry a blanket `capability,` and do not hit this). Mounting the nfsd fs
  # triggers the kernel's module autoload (kernel context, outside this label) - no sys_module/modprobe access
  # needed here.
  capability sys_admin,
  # fstype matching applies to new mounts only.
  mount fstype=(nfsd) nfsd -> /proc/fs/nfsd/,
  # Client-recovery tracker upcall channel: the nfsdcld pipe lives in the rpc_pipefs instance mounted in this
  # netns (rootfs/etc/cont-init.d/config.sh).
  mount fstype=(rpc_pipefs) sunrpc -> /var/lib/nfs/rpc_pipefs/,
  # fstype matching applies to new mounts only, so binds are conditioned on the bind flag instead.
  mount options=(bind) /{share,media}{,/**} -> /data/pseudo_root{,/**},
  umount /data/pseudo_root{,/**},
  /proc/fs/nfsd/ r,
  /proc/fs/nfsd/** rw,

  # Export tables and NFSv4 state
  /etc/exports rw,
  /etc/nfsd.args rw,
  /var/lib/nfs/** rw,
  /proc/net/rpc/** rw,

  # Network inspection (APP_IP detection) and kernel NFS state
  /proc/net/{,tcp,tcp6,if_inet6,dev} r,
  /proc/net/dev_mcast r,

  # Process introspection (ps/grep for checks, mountinfo for the nfsd-watch mirror detection)
  /proc/[0-9]*/{,cmdline,mounts,mountinfo} r,
  /proc/[0-9]*/stat r,

  # Runtime check scripts
  /etc/cont-init.d/config.sh rix,
  /etc/s6-overlay/scripts/* rix,

  ## Sub-profiles: the privileged NFS binaries run under their own tighter profiles (cx = change profile on
  ## execute). They still need the procfs/exports access listed above, hence re-granted here.

  # exportfs (nfsd-start: exportfs -a; nfsd-stop: exportfs -au)
  /usr/sbin/exportfs cx -> exportfs,

  profile exportfs flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    file,
    signal (send) set=(kill,term,int,hup,cont),
    # Official docs template: scope receive to the app's own top profile (the supervisor renames it to the
    # installed slug on the device, e.g. local_nfs-server — the *_nfs-server glob covers repo installs too).
    signal (receive) peer=*_nfs-server,

    # inet/inet6 stream+dgram: in case exportfs attempts RPC-based cache flushes toward mountd/rpcbind.
    network inet stream,
    network inet dgram,
    network inet6 stream,
    network inet6 dgram,

    /usr/sbin/exportfs rm,
    /lib/** mr,
    /etc/exports rw,
    /var/lib/nfs/** rw,
    /proc/net/rpc/** rw,
    /proc/fs/nfsd/** rw,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
    /data/** r,
  }

  # rpc.nfsd (nfsd-start: start threads; nfsd-stop: rpc.nfsd 0)
  /usr/sbin/rpc.nfsd cx -> nfsd,

  profile nfsd flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    file,
    signal (send) set=(kill,term,int,hup,cont),
    # Official docs template: scope receive to the app's own top profile (the supervisor renames it to the
    # installed slug on the device, e.g. local_nfs-server — the *_nfs-server glob covers repo installs too).
    signal (receive) peer=*_nfs-server,

    # Inert under the network deny below (no socket can exist to bind; capability mediation is independent of the
    # network class) - kept deliberately: relaxing the deny must force a fresh privileged-port decision. Facts:
    # nfs-server/KNOWLEDGE.md §"AppArmor profile — app-specific grants".
    capability net_bind_service,
    capability setgid,
    capability setuid,
    capability dac_override,

    # Network: rpc.nfsd creates NO sockets itself - it configures the kernel server via /proc/fs/nfsd/* (the
    # kernel threads own the actual NFS listeners); explicitly deny all networking (a deny-only class denies
    # everything in it; an empty class would be entirely unmediated).
    deny network,

    /usr/sbin/rpc.nfsd rm,
    /lib/** mr,
    /etc/exports rw,
    /var/lib/nfs/** rw,
    /proc/net/rpc/** rw,
    /proc/fs/nfsd/** rw,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
    /data/** r,
  }

  # rpc.mountd (mountd service; services the kernel's export-cache upcalls)
  /usr/sbin/rpc.mountd cx -> mountd,

  profile mountd flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    file,
    signal (send) set=(kill,term,int,hup,cont),
    # Official docs template: scope receive to the app's own top profile (the supervisor renames it to the
    # installed slug on the device, e.g. local_nfs-server — the *_nfs-server glob covers repo installs too).
    signal (receive) peer=*_nfs-server,

    capability net_bind_service,
    capability setgid,
    capability setuid,
    capability dac_override,

    # inet/inet6 stream+dgram: rpc.mountd is an actual RPC server (MNT service sockets TCP/UDP, v4+v6) and
    # attempts rpcbind registration (fails harmlessly without rpcbind).
    network inet stream,
    network inet dgram,
    network inet6 stream,
    network inet6 dgram,

    /usr/sbin/rpc.mountd rm,
    /lib/** mr,
    /etc/exports rw,
    /var/lib/nfs/** rw,
    /proc/net/rpc/** rw,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
    /data/** r,
  }

  # nfsdcld (cld service: client-recovery tracker daemon over the rpc_pipefs upcall pipe; sqlite store on /data/nfsdcld)
  /usr/sbin/nfsdcld cx -> nfsdcld,

  profile nfsdcld flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    file,
    signal (send) set=(kill,term,int,hup,cont),
    # Official docs template: scope receive to the app's own top profile (the supervisor renames it to the
    # installed slug on the device, e.g. local_nfs-server — the *_nfs-server glob covers repo installs too).
    signal (receive) peer=*_nfs-server,

    # setpcap: nfsdcld prunes its own capability set at startup (prctl PR_CAPBSET_DROP + a final capset); without
    # this rule the daemon exits in a restart loop ("Unable to prune capability 0 from bounding set: Operation
    # not permitted", device-enforcement-verified; the lab cannot catch this - no AppArmor enforcement there).
    # The grant only enables the self-pruning; the daemon ends up with zero capabilities anyway.
    capability setpcap,
    # Network: none at all - the daemon only talks to the kernel over the rpc_pipefs pipe (inotify + file I/O).
    deny network,

    /usr/sbin/nfsdcld rm,
    /lib/** mr,
    /data/** rw,
    /var/lib/nfs/rpc_pipefs/** rw,
    /proc/fs/nfsd/** r,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
  }
}
