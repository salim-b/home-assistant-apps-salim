#include <tunables/global>

profile nfs-server flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Generic file access, signals (s6 supervision, bash job control)
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within the app's data dir
  /data/** rw,

  # AppArmor status reporting (s6 sets this up; harmless otherwise)
  /sys/kernel/security/apparmor/** r,

  # Kernel module loading (kernel_modules: true maps /lib/modules in;
  # modprobe and ip are busybox applets, hence the /bin/busybox ix)
  /lib/modules/** r,
  /sbin/modprobe rix,
  /sbin/ip rix,
  /bin/busybox ix,

  # Mount the nfsd filesystem (SYS_ADMIN) - the app's core operation.
  # busybox `mount -t nfsd nfsd /proc/fs/nfsd` passes the device string
  # "nfsd" as the mount source, so the source cannot be constrained to a
  # path; scope by filesystem type instead. The per-share mirrors are
  # mount --bind (fstype = the underlying fs), and the down script umounts
  # them again.
  # R1 DIAGNOSTIC (temporary): blanket mount to split the problem space;
  # the scoped fstype=(nfsd) rule is loaded but the mount is still denied,
  # so the culprit may not be the rule at all. Remove once confirmed.
  mount,
  mount fstype=(nfsd),
  # NOTE: fstype matching only applies to new mounts (not bind/remount),
  # so bind mirrors are conditioned on the bind flag instead.
  mount options=(bind) /{share,media}{,/**} -> /data/pseudo_root{,/**},
  umount /data/pseudo_root{,/**},
  /proc/fs/nfsd/ r,
  /proc/fs/nfsd/** rw,

  # Export tables and NFSv4 state
  /etc/exports rw,
  /var/lib/nfs/** rw,
  /proc/net/rpc/** rw,

  # Network inspection (APP_IP detection) and kernel NFS state
  /proc/net/{,tcp,tcp6,if_inet6,dev} r,
  /proc/net/dev_mcast r,

  # Process introspection (ps/grep for checks)
  /proc/[0-9]*/{,cmdline,mounts} r,
  /proc/[0-9]*/stat r,

  # Runtime check scripts
  /etc/cont-init.d/config.sh rix,
  /etc/s6-overlay/scripts/* rix,

  ## Sub-profiles: the privileged NFS binaries run under their own tighter
  ## profiles (cx = change profile on execute). They still need the
  ## procfs/exports access listed above, hence re-granted here.

  # exportfs (runs twice: config.sh + nfsd-start/stop)
  /usr/sbin/exportfs cx -> exportfs,

  profile exportfs flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    file,
    signal (send) set=(kill,term,int,hup,cont),

    /usr/sbin/exportfs rm,
    /lib/** mr,
    /etc/exports rw,
    /var/lib/nfs/** rw,
    /proc/net/rpc/** rw,
    /proc/fs/nfsd/** rw,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
    /data/** r,
  }

  # rpc.nfsd (nfsd-start: start threads; nfsd-stop: rpc.nfsd 0)
  /usr/sbin/rpc.nfsd cx -> nfsd,

  profile nfsd flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    file,
    signal (send) set=(kill,term,int,hup,cont),

    capability net_bind_service,
    capability setgid,
    capability setuid,
    capability dac_override,

    # NOTE: no explicit network rules - AppArmor allows networking when a
    # profile has none, and rpc.nfsd itself only talks to the kernel
    # (the actual NFS sockets belong to the kernel threads, not this
    # short-lived process).

    /usr/sbin/rpc.nfsd rm,
    /lib/** mr,
    /etc/exports rw,
    /var/lib/nfs/** rw,
    /proc/net/rpc/** rw,
    /proc/fs/nfsd/** rw,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
    /data/** r,
  }

  # rpc.mountd (mountd service; services the kernel's export-cache upcalls)
  /usr/sbin/rpc.mountd cx -> mountd,

  profile mountd flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    file,
    signal (send) set=(kill,term,int,hup,cont),
    signal receive,

    capability net_bind_service,
    capability setgid,
    capability setuid,
    capability dac_override,

    /usr/sbin/rpc.mountd rm,
    /lib/** mr,
    /etc/exports rw,
    /var/lib/nfs/** rw,
    /proc/net/rpc/** rw,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
    /data/** r,
  }
}
