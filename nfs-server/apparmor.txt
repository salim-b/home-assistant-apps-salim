#include <tunables/global>

profile nfs-server flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  ## Conventions & references (device-enforcement-verified on HAOS 18.3 /
  ## OS Agent 1.14 / AppArmor 3.1.7 — see nfs-server/KNOWLEDGE.md):
  ## - shape modeled on the official dev-docs AppArmor template
  ##   (developers.home-assistant.io/docs/add-ons/presentation) and the
  ##   shipped dnsmasq app profile; sub-profile signal receive is
  ##   peer-scoped per the docs template.
  ## - `file,` blanket: upstream HA convention (docs template + dnsmasq).
  ##   Tightening file rules is impractical here because HAOS produces no
  ##   AppArmor audit/denial logs (no audit plumbing) — silent breakage
  ##   risk. The profile's confinement value sits in the non-file classes
  ##   (mount/capability), which Docker's default profile would otherwise
  ##   grant broadly (blanket `capability,`).
  ## - explicit `capability` rules are REQUIRED for privileged mounts:
  ##   AppArmor mediates capable(CAP_SYS_ADMIN) during mount(2) even when
  ##   the `mount` rule itself grants it.

  # Generic file access, signals (s6 supervision, bash job control).
  # Bare `signal,` matches the upstream dnsmasq app profile convention: it
  # grants send AND receive for all signals/peers. Cross-label signal
  # delivery needs rules on both sides (pod: sender needs send for the
  # peer's profile, receiver needs receive for the sender's); same-label
  # pairs appear to be implicitly allowed, but not relying on that is
  # cheaper than documenting it.
  file,
  signal,

  # Network mediation is enabled by the PRESENCE of any network rule; a
  # profile without network rules has networking entirely unmediated. Once
  # enabled, anything not explicitly allowed is denied - so this is a
  # close allowlist of what the supervised stack actually needs:
  # - inet/inet6 stream+dgram: bashio (supervisor API over HTTP), DNS
  #   resolution, and the Docker HEALTHCHECK's /dev/tcp probe
  # - netlink raw: busybox `ip` (rtnetlink address detection in
  #   nfsd-start / nfsd-healthcheck)
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  network netlink raw,

  # S6-Overlay
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rix,
  /run/{,**} rwk,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within the app's data dir
  /data/** rw,

  # AppArmor status reporting (s6 sets this up; harmless otherwise)
  /sys/kernel/security/apparmor/** r,

  # Address detection (APP_IP): `ip` is a busybox applet, hence the
  # /bin/busybox ix (see KNOWLEDGE.md on busybox-applet exec mediation)
  /sbin/ip rix,
  /bin/busybox ix,

  # Mount the nfsd filesystem (SYS_ADMIN) - the app's core operation.
  # busybox `mount -t nfsd nfsd /proc/fs/nfsd` passes the device string
  # "nfsd" as the mount source, so the source cannot be constrained to a
  # path; scope by filesystem type instead. The per-share mirrors are
  # mount --bind (fstype = the underlying fs), and the down script umounts
  # them again.
  # Device-iteration finding (diagnostic round): the mount mediation was
  # NOT the blocker - AppArmor also mediates capable(CAP_SYS_ADMIN) during
  # the mount syscall, and the profile lacked the capability grant (Docker's
  # default profile carries a blanket 'capability,', plain Docker containers
  # do not hit this). Mounting the nfsd fs triggers the kernel's module
  # autoload (request_module runs in kernel context, outside this label),
  # so no sys_module/modprobe access is needed here.
  capability sys_admin,
  # Tightened: the busybox mount passes the device string "nfsd" as the
  # mount source; fstype matching applies to new mounts only.
  mount fstype=(nfsd) nfsd -> /proc/fs/nfsd/,
  # NOTE: fstype matching only applies to new mounts (not bind/remount),
  # so bind mirrors are conditioned on the bind flag instead.
  mount options=(bind) /{share,media}{,/**} -> /data/pseudo_root{,/**},
  umount /data/pseudo_root{,/**},
  /proc/fs/nfsd/ r,
  /proc/fs/nfsd/** rw,

  # Export tables and NFSv4 state
  /etc/exports rw,
  /var/lib/nfs/** rw,
  /proc/net/rpc/** rw,

  # Network inspection (APP_IP detection) and kernel NFS state
  /proc/net/{,tcp,tcp6,if_inet6,dev} r,
  /proc/net/dev_mcast r,

  # Process introspection (ps/grep for checks)
  /proc/[0-9]*/{,cmdline,mounts} r,
  /proc/[0-9]*/stat r,

  # Runtime check scripts
  /etc/cont-init.d/config.sh rix,
  /etc/s6-overlay/scripts/* rix,

  ## Sub-profiles: the privileged NFS binaries run under their own tighter
  ## profiles (cx = change profile on execute). They still need the
  ## procfs/exports access listed above, hence re-granted here.

  # exportfs (runs twice: config.sh + nfsd-start/stop)
  /usr/sbin/exportfs cx -> exportfs,

  profile exportfs flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    file,
    signal (send) set=(kill,term,int,hup,cont),
    # Official docs template: scope receive to the app's own top profile
    # (the supervisor renames it to the installed slug on the device, e.g.
    # local_nfs — the *_nfs glob covers repo installs too).
    signal (receive) peer=*_nfs,

    # inet/inet6 stream+dgram: in case exportfs attempts RPC-based cache
    # flushes toward mountd/rpcbind; everything else (raw etc.) is denied
    # by the mediation-enabled allowlist model.
    network inet stream,
    network inet dgram,
    network inet6 stream,
    network inet6 dgram,

    /usr/sbin/exportfs rm,
    /lib/** mr,
    /etc/exports rw,
    /var/lib/nfs/** rw,
    /proc/net/rpc/** rw,
    /proc/fs/nfsd/** rw,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
    /data/** r,
  }

  # rpc.nfsd (nfsd-start: start threads; nfsd-stop: rpc.nfsd 0)
  /usr/sbin/rpc.nfsd cx -> nfsd,

  profile nfsd flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    file,
    signal (send) set=(kill,term,int,hup,cont),
    # Official docs template: scope receive to the app's own top profile
    # (the supervisor renames it to the installed slug on the device, e.g.
    # local_nfs — the *_nfs glob covers repo installs too).
    signal (receive) peer=*_nfs,

    capability net_bind_service,
    capability setgid,
    capability setuid,
    capability dac_override,

    # Network: rpc.nfsd creates NO sockets itself - it configures the kernel
    # server via /proc/fs/nfsd/* (the kernel threads own the actual NFS
    # listeners); explicitly deny all networking.
    # NOTE: AppArmor enables a mediation class by the PRESENCE of any rule;
    # a deny-only class denies everything in it. (An empty network class
    # would be entirely UNMEDIATED, i.e. allow all - not what we want.)
    deny network,

    /usr/sbin/rpc.nfsd rm,
    /lib/** mr,
    /etc/exports rw,
    /var/lib/nfs/** rw,
    /proc/net/rpc/** rw,
    /proc/fs/nfsd/** rw,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
    /data/** r,
  }

  # rpc.mountd (mountd service; services the kernel's export-cache upcalls)
  /usr/sbin/rpc.mountd cx -> mountd,

  profile mountd flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    file,
    signal (send) set=(kill,term,int,hup,cont),
    # Official docs template: scope receive to the app's own top profile
    # (the supervisor renames it to the installed slug on the device, e.g.
    # local_nfs — the *_nfs glob covers repo installs too).
    signal (receive) peer=*_nfs,

    capability net_bind_service,
    capability setgid,
    capability setuid,
    capability dac_override,

    # inet/inet6 stream+dgram: rpc.mountd is an actual RPC server - it
    # creates its MNT service sockets (TCP/UDP, v4+v6) and attempts rpcbind
    # registration (fails harmlessly without rpcbind, same as unmediated);
    # everything else (raw etc.) is denied by the allowlist model.
    network inet stream,
    network inet dgram,
    network inet6 stream,
    network inet6 dgram,

    /usr/sbin/rpc.mountd rm,
    /lib/** mr,
    /etc/exports rw,
    /var/lib/nfs/** rw,
    /proc/net/rpc/** rw,
    /proc/[0-9]*/{,cmdline,mounts} r,
    /tmp/** rwk,
    /data/** r,
  }
}
