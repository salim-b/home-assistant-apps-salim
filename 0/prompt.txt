New classify(old_path, new_path, out_path) groups file<TAB>slug<TAB>ref detail lines per (file, slug) and emits one etype<TAB>slug<TAB>old<TAB>new row per actual transition; identical transitions across files deduplicated via dict.fromkeys; identical-to-other-side keys skipped; single-ref replacement → review/nondigest by DIGEST.fullmatch; multi-vs-multi → ambiguous; added rows use '-' placeholder for empty side.
```
def classify(old_path: str, new_path: str, out_path: str) -> None:
    def load(path: str) -> dict[tuple[str, str], set[str]]:
        pins: dict[tuple[str, str], set[str]] = {}
        for line in Path(path).read_text(encoding="utf-8").splitlines():
            if not line: continue
            file, slug, ref = line.split("\t", 2)
            pins.setdefault((file, slug), set()).add(ref)
        return pins
    old, new = load(old_path), load(new_path)
    rows: list[tuple[str, str, str, str]] = []
    for key in sorted(set(old) | set(new)):
        slug = key[1]
        o, n = old.get(key, set()), new.get(key, set())
        if o == n: continue
        if not o: rows.append(("added", slug, "-", " ".join(sorted(n))))
        elif not n: rows.append(("removed", slug, " ".join(sorted(o)), "-"))
        elif len(o) == 1 and len(n) == 1:
            ov, nv = min(o), min(n)
            etype = "review" if DIGEST.fullmatch(ov) and DIGEST.fullmatch(nv) else "nondigest"
            rows.append((etype, slug, ov, nv))
        else:
            rows.append(("ambiguous", slug, " ".join(sorted(o)), " ".join(sorted(n))))
    deduped = list(dict.fromkeys(rows))
    Path(out_path).write_text("".join("\t".join(row) + "\n" for row in deduped), encoding="utf-8")
```
